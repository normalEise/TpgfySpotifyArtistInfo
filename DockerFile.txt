# --- Stage 1: Build the application ---
FROM maven:3.9-eclipse-temurin-17 AS build
WORKDIR /app

# Copy the pom.xml and source code
COPY spotify-api/pom.xml ./spotify-api/
COPY spotify-api/src ./spotify-api/src

# Build the application from inside the spotify-api folder
WORKDIR /app/spotify-api
RUN mvn clean package -DskipTests

# --- Stage 2: Run the application ---
FROM eclipse-temurin:17-jre-alpine
WORKDIR /app

# Copy the built jar from the build stage
COPY --from=build /app/spotify-api/target/*.jar app.jar

# Spring Boot default port
EXPOSE 8080

# Run the app
ENTRYPOINT ["java", "-jar", "app.jar"]
